===== 9. Ψ 加權 trainer：train_step 完整內容（含行號）=====
 121      def train_step(self, batch: dict) -> dict:
 122          data = batch['data']
 123          target = batch['target']
 124          keys = batch.get('keys')
 125  
 126          data = data.to(self.device, non_blocking=True)
 127          if isinstance(target, list):
 128              target = [i.to(self.device, non_blocking=True) for i in target]
 129          else:
 130              target = target.to(self.device, non_blocking=True)
 131  
 132          n = data.shape[0]
 133          w = self._weights_for(keys, n).to(self.device)
 134  
 135          self.optimizer.zero_grad(set_to_none=True)
 136          with autocast(self.device.type, enabled=True) if self.device.type == 'cuda' else dummy_context():
 137              output = self.network(data)
 138  
 139              if torch.allclose(w, torch.ones_like(w)):
 140                  # 這個 batch 沒有任何加權樣本，走原始路徑（較快且數值一致）
 141                  l = self.loss(output, target)
 142              else:
 143                  # 逐樣本算 loss 後加權平均，分母為權重和以維持尺度
 144                  total = 0.0
 145                  for i in range(n):
 146                      li = self.loss(self._sample_slice(output, i),
 147                                     self._sample_slice(target, i))
 148                      total = total + w[i] * li
 149                  l = total / w.sum()
 150  
 151          if self.grad_scaler is not None:
 152              self.grad_scaler.scale(l).backward()
 153              self.grad_scaler.unscale_(self.optimizer)
 154              torch.nn.utils.clip_grad_norm_(self.network.parameters(), 12)
 155              self.grad_scaler.step(self.optimizer)
 156              self.grad_scaler.update()
 157          else:
 158              l.backward()
 159              torch.nn.utils.clip_grad_norm_(self.network.parameters(), 12)
 160              self.optimizer.step()
 161  
 162          return {'loss': l.detach().cpu().numpy()}
